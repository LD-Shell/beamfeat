# Pinned to the environment in which the reported results were produced.
# Two independent ceilings make a dedicated environment mandatory for autofeat:
#   * scikit-learn MUST be < 1.8: autofeat 2.1.3 crashes on >= 1.8
#     (check_array force_all_finite removal).
#   * numpy MUST be <= 2.2: autofeat pulls in numba, which refuses to import
#     above that ("Numba needs NumPy 2.2 or less"). Installing autofeat into
#     an environment with a newer numpy fails at import on every fit.
numpy==1.26.4
scipy>=1.11
pandas>=2.0
scikit-learn==1.7.2
lightgbm==4.7.0
autofeat==2.1.3
openfe==0.0.12
featuretools==1.31.0
# featuretools 1.31 imports pkg_resources, which setuptools removed in 82.0.
# Without this pin every featuretools fit fails with ModuleNotFoundError.
setuptools<82
pyarrow
matplotlib

# Selection-only baseline for benchmarks/run_benchmarks.py --methods knockpy.
# Unconstrained, knockpy pulls a cvxpy build that requires numpy >= 2, which
# breaks autofeat and numba. Listed after the numpy pin so the resolver is
# forced to an older, compatible pair (knockpy 1.3.2 with cvxpy 1.7.5).
# Install with `pip install -r requirements.txt`, never `pip install knockpy`
# into an already-built environment.
knockpy
